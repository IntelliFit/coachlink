## 🔍 API接口检查（route.ts）

### ✅ 必须检查项

#### 1. 导入语句
```typescript
// ✅ 正确
import { NextResponse } from 'next/server';
import { getServerSession } from '../../auth/[...nextauth]/auth';
import supabase from '@/lib/supabaseClient';
import type { RequestDto, ResponseDto } from './entity';

// ❌ 错误：缺少必要的导入
import { NextResponse } from 'next/server';
// 缺少 getServerSession 和 supabase
```

#### 2. 登录检查
```typescript
// ✅ 正确
const session = await getServerSession();
if (!session?.user?.id) {
    return NextResponse.json({
        success: false,
        message: '用户未登录',
        data: null
    }, { status: 401 });
}

// ❌ 错误：没有检查登录
export async function POST(req: Request) {
    // 直接处理请求，没有检查登录
}
```

#### 3. 获取用户ID
```typescript
// ✅ 正确：从session获取
const userId = session.user.id;

// ❌ 错误：从请求参数获取
const { userId } = await req.json(); // 不安全！
```

#### 4. 返回格式
```typescript
// ✅ 正确：统一格式
return NextResponse.json<ApiData<ResponseDto>>({
    success: true,
    message: '操作成功',
    data: result
});

// ❌ 错误：格式不一致
return NextResponse.json({ result }); // 缺少success和message
return NextResponse.json({ data: result }); // 缺少success和message
```

#### 5. 错误处理
```typescript
// ✅ 正确：检查数据库错误
const { data, error } = await supabase.from('users').select('*');
if (error) {
    return NextResponse.json({
        success: false,
        message: '查询失败',
        data: null
    });
}

// ❌ 错误：没有检查error
const { data } = await supabase.from('users').select('*');
return NextResponse.json({ success: true, data }); // 可能data是null
```

## 🔍 Entity文件检查（entity.ts）

### ✅ 必须检查项

#### 1. 请求参数类型
```typescript
// ✅ 正确：通常不需要userId（从session获取）
export interface GetUserProfileDto {
    // 空对象或只有必要的查询参数
}

// ⚠️ 注意：如果确实需要userId，应该有说明
export interface GetUserProfileDto {
    userId?: string; // 可选，通常从session获取
}

// ❌ 错误：不应该要求前端传userId
export interface GetUserProfileDto {
    userId: string; // 应该从session获取
}
```

#### 2. 返回数据类型
```typescript
// ✅ 正确：包含所有必要字段
export interface GetUserProfileResponseDto {
    id: string;
    name: string;
    email: string;
    avatar_url?: string; // 可选字段用?
}

// ❌ 错误：缺少必要字段
export interface GetUserProfileResponseDto {
    name: string; // 缺少id和email
}
```

## 🔍 Service层检查（service/*.ts）

### ✅ 必须检查项

#### 1. 继承BaseService
```typescript
// ✅ 正确
import { BaseService } from "@/lib/baseService";

export class UserService extends BaseService {
    // ...
}

// ❌ 错误：没有继承
export class UserService {
    // 无法使用this.post()等方法
}
```

#### 2. API调用方法
```typescript
// ✅ 正确：使用this.post()或this.get()
async getUserProfile(reqData: GetUserProfileDto) {
    const res = await this.post<GetUserProfileResponseDto>('user/get-profile', reqData);
    if (res.data && res.data.success) {
        return res.data.data || null;
    }
    return null;
}

// ❌ 错误：直接使用fetch
async getUserProfile(reqData: GetUserProfileDto) {
    const res = await fetch('/api/user/get-profile', {
        method: 'POST',
        body: JSON.stringify(reqData)
    });
    // 没有统一处理
}
```

#### 3. 成功状态检查
```typescript
// ✅ 正确：检查success
if (res.data && res.data.success) {
    return res.data.data || null;
}

// ❌ 错误：没有检查success
return res.data.data; // 可能失败时也返回了
```

## 🔍 前端组件检查（page.tsx）

### ✅ 必须检查项

#### 1. 客户端组件标记
```typescript
// ✅ 正确：需要交互的组件
'use client';

import { useState, useEffect } from 'react';

// ❌ 错误：需要交互但没有标记
import { useState } from 'react'; // 会报错
```

#### 2. 使用Service层
```typescript
// ✅ 正确：通过Service调用
import { UserService } from '@/app/service/userService';

const data = await new UserService().getUserProfile({});

// ❌ 错误：直接调用API或访问数据库
const res = await fetch('/api/user/get-profile');
// 或
const { data } = await supabase.from('users').select('*');
```

#### 3. 错误提示
```typescript
// ✅ 正确：使用toast
import { toast } from '@/context/ToastContext';

try {
    await service.doSomething();
} catch (error) {
    toast.error('操作失败');
}

// ❌ 错误：使用setError
const [error, setError] = useState('');
setError('操作失败'); // 应该用toast
```

#### 4. Loading状态
```typescript
// ✅ 正确：使用骨架屏
import { Skeleton } from 'primereact/skeleton';

if (loading) {
    return <Skeleton height="2rem" />;
}

// ❌ 错误：简单的文字提示
if (loading) {
    return <div>加载中...</div>; // 应该用骨架屏
}
```

#### 5. 移动端适配
```typescript
// ✅ 正确：响应式设计
<div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3">
    {/* 内容 */}
</div>

// ❌ 错误：固定宽度
<div className="w-96">
    {/* 在小屏幕上会溢出 */}
</div>
```

#### 6. 使用PrimeReact组件
```typescript
// ✅ 正确：使用PrimeReact组件
import { Button } from 'primereact/button';
import { DataTable } from 'primereact/datatable';

<Button label="提交" onClick={handleSubmit} />
<DataTable value={data} />

// ❌ 错误：使用原生HTML
<button onClick={handleSubmit}>提交</button>
<table>
    {/* 应该用DataTable */}
</table>
```

## 🔍 数据库操作检查

### ✅ 必须检查项

#### 1. 表名正确性
```typescript
// ✅ 正确：参考supabase_schema.md
await supabase.from('coach_plan_templates').select('*');
await supabase.from('plan_instance_executions').select('*');

// ❌ 错误：表名不存在或拼写错误
await supabase.from('coach_templates').select('*'); // 应该是coach_plan_templates
```

#### 2. 错误检查
```typescript
// ✅ 正确：检查error
const { data, error } = await supabase.from('users').select('*');
if (error) {
    console.error('查询失败:', error);
    return NextResponse.json({
        success: false,
        message: '查询失败',
        data: null
    });
}

// ❌ 错误：没有检查error
const { data } = await supabase.from('users').select('*');
// 如果出错，data可能是null，但没有处理
```

#### 3. 查询条件
```typescript
// ✅ 正确：使用正确的字段名
const { data } = await supabase
    .from('coach_plan_instances')
    .select('*')
    .eq('student_id', userId); // 字段名正确

// ❌ 错误：字段名错误
const { data } = await supabase
    .from('coach_plan_instances')
    .select('*')
    .eq('userId', userId); // 应该是student_id
```

#### 4. 权限检查
```typescript
// ✅ 正确：检查用户权限
const session = await getServerSession();
if (!session?.user?.id) {
    return NextResponse.json({
        success: false,
        message: '未登录',
        data: null
    }, { status: 401 });
}

// 检查是否是教练
if (!session?.user?.is_coach) {
    return NextResponse.json({
        success: false,
        message: '无权限',
        data: null
    }, { status: 403 });
}

// ❌ 错误：没有权限检查
const { data } = await supabase
    .from('coaches')
    .select('*'); // 任何人都能访问
```

## 🔍 常见错误模式

### ❌ 错误1：前端直接访问数据库
```typescript
// ❌ 错误
'use client';
import supabase from '@/lib/supabaseClient';

const { data } = await supabase.from('users').select('*');

// ✅ 正确
import { UserService } from '@/app/service/userService';
const data = await new UserService().getUsers();
```

### ❌ 错误2：从请求参数获取userId
```typescript
// ❌ 错误
export async function POST(req: Request) {
    const { userId } = await req.json();
    // 不安全，用户可以伪造userId
}

// ✅ 正确
export async function POST(req: Request) {
    const session = await getServerSession();
    const userId = session.user.id;
}
```

### ❌ 错误3：返回格式不一致
```typescript
// ❌ 错误
return NextResponse.json({ data: result });
return NextResponse.json({ result });

// ✅ 正确
return NextResponse.json({
    success: true,
    message: '操作成功',
    data: result
});
```

### ❌ 错误4：没有检查数据库错误
```typescript
// ❌ 错误
const { data } = await supabase.from('users').select('*');
return NextResponse.json({ success: true, data });

// ✅ 正确
const { data, error } = await supabase.from('users').select('*');
if (error) {
    return NextResponse.json({
        success: false,
        message: '查询失败',
        data: null
    });
}
return NextResponse.json({ success: true, data });
```

### ❌ 错误5：使用setError而不是toast
```typescript
// ❌ 错误
const [error, setError] = useState('');
setError('操作失败');

// ✅ 正确
import { toast } from '@/context/ToastContext';
toast.error('操作失败');
```

## 📋 快速检查表

复制这个表格，检查代码时打勾：

### API接口（route.ts）
- [ ] 导入了NextResponse、getServerSession、supabase
- [ ] 检查了登录状态（session?.user?.id）
- [ ] 从session获取userId（不是从请求参数）
- [ ] 返回格式为`{success, message, data}`
- [ ] 检查了数据库操作的error
- [ ] 使用了类型定义（import type）

### Entity文件（entity.ts）
- [ ] 请求参数类型合理（通常不需要userId）
- [ ] 返回数据类型包含所有必要字段
- [ ] 可选字段使用了`?`

### Service层（service/*.ts）
- [ ] 继承了BaseService
- [ ] 使用了this.post()或this.get()
- [ ] 检查了res.data.success
- [ ] 返回了res.data.data或null

### 前端组件（page.tsx）
- [ ] 需要交互的组件标记了'use client'
- [ ] 使用了Service层（不是直接调用API）
- [ ] 使用了toast显示错误（不是setError）
- [ ] Loading时使用了骨架屏（不是文字）
- [ ] 适配了移动端（响应式设计）
- [ ] 使用了PrimeReact组件（不是原生HTML）

### 数据库操作
- [ ] 表名正确（参考supabase_schema.md）
- [ ] 检查了error
- [ ] 查询条件使用了正确的字段名
- [ ] 有适当的权限检查

## 🎯 检查优先级

### 🔴 高优先级（必须修复）
1. 前端直接访问数据库
2. 从请求参数获取userId
3. 没有检查登录状态
4. 返回格式不一致

### 🟡 中优先级（建议修复）
1. 没有检查数据库error
2. 使用setError而不是toast
3. Loading没有用骨架屏
4. 没有适配移动端

### 🟢 低优先级（可以优化）
1. 类型定义不完整
2. 代码注释不足
3. 变量命名不规范
