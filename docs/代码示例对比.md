### 示例1：获取用户信息

#### ❌ 错误示例
```typescript
// src/app/api/user/get-profile/route.ts

import { NextResponse } from 'next/server';

export async function POST(req: Request) {
    // ❌ 错误1：没有检查登录
    // ❌ 错误2：从请求参数获取userId（不安全）
    const { userId } = await req.json();
    
    // ❌ 错误3：前端直接访问数据库（应该在API中）
    // ❌ 错误4：没有检查error
    const { data } = await supabase
        .from('users')
        .select('*')
        .eq('id', userId)
        .single();
    
    // ❌ 错误5：返回格式不一致
    return NextResponse.json({ user: data });
}
```

#### ✅ 正确示例
```typescript
// src/app/api/user/get-profile/route.ts

import { NextResponse } from 'next/server';
import { getServerSession } from '../../auth/[...nextauth]/auth';
import supabase from '@/lib/supabaseClient';
import type { GetUserProfileDto, GetUserProfileResponseDto } from './entity';
import type { ApiData } from '@/types/api';

export async function POST(req: Request) {
    // ✅ 正确1：检查登录状态
    const session = await getServerSession();
    if (!session?.user?.id) {
        return NextResponse.json({
            success: false,
            message: '用户未登录',
            data: null
        }, { status: 401 });
    }

    // ✅ 正确2：从session获取userId（安全）
    const userId = session.user.id;

    // ✅ 正确3：在API中访问数据库
    // ✅ 正确4：检查error
    const { data, error } = await supabase
        .from('user_profiles')
        .select('*')
        .eq('user_id', userId)
        .single();

    if (error) {
        return NextResponse.json({
            success: false,
            message: '查询失败',
            data: null
        });
    }

    // ✅ 正确5：统一返回格式
    return NextResponse.json<ApiData<GetUserProfileResponseDto>>({
        success: true,
        message: '获取成功',
        data: data
    });
}
```

```typescript
// src/app/api/plan-lib/create-template/route.ts

import { NextResponse } from 'next/server';
import { getServerSession } from '../../auth/[...nextauth]/auth';
import supabase from '@/lib/supabaseClient';
import type { CreateTemplateDto, CreateTemplateResponseDto } from './entity';
import type { ApiData } from '@/types/api';

export async function POST(req: Request) {
    // ✅ 正确1：检查登录
    const session = await getServerSession();
    if (!session?.user?.id) {
        return NextResponse.json({
            success: false,
            message: '用户未登录',
            data: null
        }, { status: 401 });
    }

    // ✅ 正确2：检查是否是教练
    if (!session?.user?.is_coach) {
        return NextResponse.json({
            success: false,
            message: '无权限，只有教练可以创建模板',
            data: null
        }, { status: 403 });
    }

    // ✅ 正确3：从session获取coachId
    const coachId = session.user.id;
    const { name, training_days } = await req.json() as CreateTemplateDto;

    // ✅ 正确4：检查error
    const { data, error } = await supabase
        .from('coach_plan_templates')
        .insert({
            coach_id: coachId,
            name: name,
            training_days: training_days
        })
        .select()
        .single();

    if (error) {
        return NextResponse.json({
            success: false,
            message: '创建失败：' + error.message,
            data: null
        });
    }

    // ✅ 正确5：统一返回格式
    return NextResponse.json<ApiData<CreateTemplateResponseDto>>({
        success: true,
        message: '创建成功',
        data: data
    });
}
```

#### ✅ 正确示例

```typescript
// src/app/service/userService.ts

import { BaseService } from "@/lib/baseService";
import type { GetUserProfileDto, GetUserProfileResponseDto } from '@/app/api/user/get-profile/entity';

export class UserService extends BaseService {
    // ✅ 正确1：继承BaseService
    // ✅ 正确2：使用this.post()方法
    async getUserProfile(reqData: GetUserProfileDto = {}) {
        const res = await this.post<GetUserProfileResponseDto>('user/get-profile', reqData);
        
        // ✅ 正确3：检查success
        if (res.data && res.data.success) {
            return res.data.data || null;
        }
        return null;
    }
}
```

```typescript
// src/app/dashboard/profile/page.tsx

'use client'; // ✅ 正确1：标记客户端组件

import { useState, useEffect } from 'react';
import { useSession } from 'next-auth/react';
import { Skeleton } from 'primereact/skeleton';
import { Card } from 'primereact/card';
import { toast } from '@/context/ToastContext';
import { UserService } from '@/app/service/userService';

export default function ProfilePage() {
    const { data: session } = useSession();
    const [profile, setProfile] = useState(null);
    const [loading, setLoading] = useState(true);

    useEffect(() => {
        // ✅ 正确2：使用Service层
        async function loadProfile() {
            try {
                const data = await new UserService().getUserProfile({});
                if (data) {
                    setProfile(data);
                } else {
                    toast.error('加载失败'); // ✅ 正确3：使用toast
                }
            } catch (error) {
                toast.error('网络错误');
            } finally {
                setLoading(false);
            }
        }
        loadProfile();
    }, []);

    // ✅ 正确4：Loading用骨架屏
    if (loading) {
        return (
            <div className="p-4">
                <Skeleton height="2rem" className="mb-2" />
                <Skeleton height="10rem" />
            </div>
        );
    }

    return (
        <div className="p-4">
            {/* ✅ 正确5：使用PrimeReact组件 */}
            <Card title="用户资料" className="max-w-2xl mx-auto">
                <div className="space-y-4">
                    <div>
                        <label className="text-surface-600">姓名</label>
                        <p className="text-surface-900">{profile?.full_name || '未设置'}</p>
                    </div>
                </div>
            </Card>
        </div>
    );
}
```

## 📋 总结

### 关键原则

1. **安全第一**：永远从session获取userId，不要从请求参数获取
2. **统一格式**：API返回格式必须是`{success, message, data}`
3. **错误处理**：所有数据库操作都要检查error
4. **分层架构**：前端 → Service → API → 数据库
5. **用户体验**：使用toast显示错误，使用骨架屏显示Loading

### 快速识别错误

- ❌ 看到`import supabase`在前端组件中 → 错误
- ❌ 看到`const { userId } = await req.json()` → 错误
- ❌ 看到`setError()` → 应该用`toast.error()`
- ❌ 看到`<div>加载中...</div>` → 应该用`<Skeleton />`
- ❌ 看到`<button>` → 应该用`<Button>`（PrimeReact）
- ❌ 看到返回格式不是`{success, message, data}` → 错误
